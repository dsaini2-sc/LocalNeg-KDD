\section{Embeddings Already Know the Graph}
\label{sec:locality}
% Observation 1. Measured on a model trained with random negatives only.

\begin{figure}[t]
  \centering
  \includegraphics[width=0.85\columnwidth]{figures/locality}
  \caption{Random-trained embeddings are graph-local. Mean cosine similarity in the query embedding space between 20{,}000 users and targets at exactly 1, 2 and 3 hops, random users, and the users' own future friends (dashed). Two random-negative models trained from different seeds; 95\% intervals are narrower than the line width.}
  \Description{Bar chart: cosine similarity falls from about 0.70 at one hop to 0.62 at two hops, 0.45 at three hops and 0.07 for random users; future friends sit at 0.61.}
  \label{fig:locality}
\end{figure}
