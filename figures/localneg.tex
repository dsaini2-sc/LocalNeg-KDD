% LocalNeg method diagram: (a) one window, GPU training and CPU planning overlapped; (b) random vs clustered batch;
% (c) GPU time per window for the three ways of batching. Widths in (c) follow Table 2 (4.80 / 4.10 + 62 s / 4.01 s).
\begin{tikzpicture}[
    font=\sffamily\scriptsize, >=Stealth, node distance=4mm,
    gpu/.style={draw=blue!60!black, fill=blue!8, rounded corners=2pt, align=center, minimum height=9mm, inner sep=2pt},
    cpu/.style={draw=orange!70!black, fill=orange!10, rounded corners=2pt, align=center, minimum height=9mm, inner sep=2pt},
    store/.style={draw=gray!70!black, fill=gray!10, cylinder, shape border rotate=90, aspect=0.2, align=center,
      minimum height=8mm, minimum width=16mm, inner sep=1pt},
    lane/.style={font=\sffamily\scriptsize\bfseries, anchor=east},
    flow/.style={->, thick, gray!50!black},
    panel/.style={font=\sffamily\footnotesize\bfseries, anchor=north west}]

  % (a) one window
  \node[panel] at (-1.1, 2.3) {(a) One \ours{} window};
  \node[lane, blue!60!black] at (-0.05, 1.3) {GPU};
  \node[lane, orange!70!black] at (-0.05, -0.95) {CPU};
  \node[gpu, text width=30mm] (train) at (1.55, 1.3) {Train window $t$:\\50 updates of 2{,}048 rows};
  \node[store] (cache) at (1.55, 0.05) {};
  \node[font=\sffamily\tiny, align=center] at (cache.center) {query-embedding\\cache};
  \node[cpu, text width=29mm] (cluster) at (4.55, -0.95) {Cluster window $t{+}1$'s\\anchors, $\sim$64 per cluster};
  \node[cpu, text width=29mm] (pack) at (7.75, -0.95) {Pack batches: 16 clusters,\\\rowcap{}: $\le$128 rows each};
  \node[gpu, text width=29mm] (next) at (7.75, 1.3) {Train window $t{+}1$};
  \draw[flow] (train.south) -- (cache.north);
  \node[font=\sffamily\tiny, align=left, anchor=west] at (1.7, 0.66) {anchors' query embeddings\\(a by-product of training)};
  \draw[flow] (cache.south) |- node[pos=0.75, above, font=\sffamily\tiny] {snapshot} (cluster.west);
  \draw[flow] (cluster.east) -- (pack.west);
  \draw[flow] (pack.north) -- (next.south);
  \draw[flow, dashed] (train.east) -- node[above, font=\sffamily\tiny, align=center] {GPU keeps training\\while the CPU plans} (next.west);

  % (b) random vs clustered batch
  \begin{scope}[shift={(10.0, -1.0)}]
    \node[panel] at (-0.25, 3.3) {(b) Who shares a batch};
    \foreach \x/\ttl in {0/Random batch, 2.25/\ours{} batch} {
      \draw[gray!60, rounded corners=2pt] (\x, 0) rectangle ++(2.0, 2.5);
      \node[font=\sffamily\scriptsize] at (\x + 1.0, 2.7) {\ttl};
    }
    \pgfmathsetseed{7}
    \foreach \i in {1,...,40} {
      \fill[blue!60!black] (0.12 + 1.76 * rnd, 0.12 + 2.26 * rnd) circle (0.9pt);
    }
    \foreach \cx/\cy/\c in {2.75/1.85/blue!60!black, 3.65/1.95/orange!80!black, 2.85/0.65/green!50!black,
        3.7/0.7/red!70!black} {
      \draw[\c, dotted] (\cx, \cy) circle (0.36);
      \foreach \i in {1,...,10} {
        \fill[\c] (\cx - 0.28 + 0.56 * rnd, \cy - 0.28 + 0.56 * rnd) circle (0.9pt);
      }
    }
    \node[font=\sffamily\tiny, align=center] at (1.0, -0.3) {negatives far away,\\many sampled nodes};
    \node[font=\sffamily\tiny, align=center] at (3.25, -0.3) {negatives from the same region,\\fewer sampled nodes};
  \end{scope}

  % (c) GPU time per window
  \begin{scope}[shift={(0, -2.15)}, yscale=0.42]
    \node[panel] at (-1.1, 0.95) {(c) GPU time over three 50-update windows};
    \foreach \y/\lbl/\step/\inf in {0/Random/4.80/0, -1/Fresh signal/4.10/62, -2/\ours{}/4.01/0} {
      \node[lane, black] at (-0.15, \y - 0.5) {\lbl};
      \pgfmathsetmacro{\w}{50 * \step / 100}
      \pgfmathsetmacro{\v}{\inf / 100}
      \foreach \k in {0, 1, 2} {
        \pgfmathsetmacro{\x}{\k * (\w + \v)}
        \ifdim \v pt > 0pt
          \fill[red!45, draw=red!60!black] (\x, \y - 0.85) rectangle ++(\v, 0.7);
        \fi
        \fill[blue!12, draw=blue!60!black] (\x + \v, \y - 0.85) rectangle ++(\w, 0.7);
      }
    }
    \node[font=\sffamily\tiny, red!60!black, anchor=west] at (8.15, -1.5) {re-embedding on the critical path};
    \node[font=\sffamily\tiny, orange!70!black, anchor=west] at (6.2, -2.7) {clustering overlapped on the CPU};
    \foreach \k in {0, 1, 2} {
      \fill[orange!35, draw=orange!70!black] (\k * 2.005 + 0.15, -3.15) rectangle ++(0.5, 0.25);
    }
  \end{scope}
\end{tikzpicture}
